## PART1
### 表结构
+ 套利扫描表 arbitrage_scan
```sql
CREATE TABLE arbitrage_scan (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    
    wallet_address VARCHAR(64) NOT NULL,
    wallet_sol_balance DECIMAL(20,9),
    
    scan_start_time DATETIME(6),
    scan_end_time DATETIME(6),
    
    scanned_paths INT DEFAULT 0,
    found_opportunities INT DEFAULT 0,
    
    created_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6)
);
```
+ 套利机会表 arbitrage_opportunity
```sql
CREATE TABLE arbitrage_opportunity (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    
    scan_id BIGINT NOT NULL,
    
    path VARCHAR(100) NOT NULL, 
    -- 例如: USDC->SOL->USD1->USDC
    
    input_token VARCHAR(20),
    input_amount DECIMAL(30,12),
    
    expected_output_amount DECIMAL(30,12),
    expected_profit DECIMAL(30,12),
    expected_profit_bps INT,
    
    min_profit_threshold_bps INT,
    
    status VARCHAR(20) DEFAULT 'SKIPPED',
    -- SKIPPED / EXECUTED
    
    created_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6),
    
    INDEX idx_scan_id (scan_id)
);
```
+ 套利执行记录 arbitrage_execution
```sql
CREATE TABLE arbitrage_execution (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    
    opportunity_id BIGINT NOT NULL,
    
    start_time DATETIME(6),
    end_time DATETIME(6),
    
    actual_output_amount DECIMAL(30,12),
    actual_profit DECIMAL(30,12),
    actual_profit_bps INT,
    
    gas_fee_sol DECIMAL(20,9),
    
    status VARCHAR(20),
    -- SUCCESS / FAILED
    
    error_message TEXT,
    
    created_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6),
    
    INDEX idx_opportunity_id (opportunity_id)
);
```
+ 三步交易记录表 arbitrage_tx
```sql
CREATE TABLE arbitrage_tx (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    
    execution_id BIGINT NOT NULL,
    
    step_number INT, 
    -- 1 / 2 / 3
    
    from_token VARCHAR(20),
    to_token VARCHAR(20),
    
    input_amount DECIMAL(30,12),
    output_amount DECIMAL(30,12),
    
    tx_signature VARCHAR(128),
    
    status VARCHAR(20),
    -- SUCCESS / FAILED
    
    created_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6),
    
    INDEX idx_execution_id (execution_id)
);
```
